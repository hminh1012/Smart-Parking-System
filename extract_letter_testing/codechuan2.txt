// =============================================
// HÀM TIỆN ÍCH
// =============================================
uint8_t getPixel(int x, int y) {
    if(x < 0 || x >= IMG_WIDTH || y < 0 || y >= IMG_HEIGHT) return 0;
    return psram_img[y * IMG_WIDTH + x];
}

bool isBlackPixel(int x, int y) {
    return getPixel(x, y) < 97;
}

bool isWhitePixel(int x, int y) {
    return getPixel(x, y) > 130;
}

int min(int a, int b) { return (a < b) ? a : b; }
int max(int a, int b) { return (a > b) ? a : b; }

// CROP VIỀN THỰC TẾ
// =============================================
void deepCropBorder() {
    Serial.println("\n=== XÓA VIỀN SÂU ===");
    
    const int REQUIRED_CONFIDENCE = 70;
    const int SCAN_STEP = 2;
    const int EXPAND_MARGIN = 3;
    
    // VIỀN TRÁI
    Serial.println("\n1. XÓA SÂU VIỀN TRÁI:");
    plateStartX = 0;
    for(int x = 0; x < IMG_WIDTH; x += SCAN_STEP) {
        int whiteCount = 0;
        int sampledPixels = 0;
        
        for(int y = 0; y < IMG_HEIGHT; y += 2) {
            if(isWhitePixel(x, y)) whiteCount++;
            sampledPixels++;
        }
        
        float whiteRatio = (whiteCount * 100.0) / sampledPixels;
        
        if(whiteRatio >= REQUIRED_CONFIDENCE) {
            bool confirmed = true;
            for(int dx = x + 1; dx <= min(IMG_WIDTH-1, x + 3); dx++) {
                int nextWhiteCount = 0;
                for(int y = 0; y < IMG_HEIGHT; y += 4) {
                    if(isWhitePixel(dx, y)) nextWhiteCount++;
                }
                float nextRatio = (nextWhiteCount * 100.0) / (IMG_HEIGHT / 4);
                if(nextRatio < 60) {
                    confirmed = false;
                    break;
                }
            }
            
            if(confirmed) {
                plateStartX = min(IMG_WIDTH-1, x + EXPAND_MARGIN);
                Serial.printf("   → Biên trái tại X=%d (tỷ lệ trắng: %.1f%%)\n", plateStartX, whiteRatio);
                break;
            }
        }
    }
    
    // VIỀN PHẢI
    Serial.println("\n2. XÓA SÂU VIỀN PHẢI:");
    plateEndX = IMG_WIDTH - 1;
    for(int x = IMG_WIDTH - 1; x >= 0; x -= SCAN_STEP) {
        int whiteCount = 0;
        int sampledPixels = 0;
        
        for(int y = 0; y < IMG_HEIGHT; y += 2) {
            if(isWhitePixel(x, y)) whiteCount++;
            sampledPixels++;
        }
        
        float whiteRatio = (whiteCount * 100.0) / sampledPixels;
        
        if(whiteRatio >= REQUIRED_CONFIDENCE) {
            bool confirmed = true;
            for(int dx = x - 1; dx >= max(0, x - 3); dx--) {
                int nextWhiteCount = 0;
                for(int y = 0; y < IMG_HEIGHT; y += 4) {
                    if(isWhitePixel(dx, y)) nextWhiteCount++;
                }
                float nextRatio = (nextWhiteCount * 100.0) / (IMG_HEIGHT / 4);
                if(nextRatio < 60) {
                    confirmed = false;
                    break;
                }
            }
            
            if(confirmed) {
                plateEndX = max(0, x - EXPAND_MARGIN);
                Serial.printf("   → Biên phải tại X=%d (tỷ lệ trắng: %.1f%%)\n", plateEndX, whiteRatio);
                break;
            }
        }
    }
    
    // VIỀN TRÊN
    Serial.println("\n3. XÓA SÂU VIỀN TRÊN:");
    plateStartY = 0;
    for(int y = 0; y < IMG_HEIGHT; y += SCAN_STEP) {
        int whiteCount = 0;
        int sampledPixels = 0;
        
        for(int x = plateStartX; x <= plateEndX; x += 2) {
            if(isWhitePixel(x, y)) whiteCount++;
            sampledPixels++;
        }
        
        float whiteRatio = (whiteCount * 100.0) / sampledPixels;
        
        if(whiteRatio >= REQUIRED_CONFIDENCE) {
            bool confirmed = true;
            for(int dy = y + 1; dy <= min(IMG_HEIGHT-1, y + 3); dy++) {
                int nextWhiteCount = 0;
                for(int x = plateStartX; x <= plateEndX; x += 4) {
                    if(isWhitePixel(x, dy)) nextWhiteCount++;
                }
                float nextRatio = (nextWhiteCount * 100.0) / ((plateEndX - plateStartX + 1) / 4);
                if(nextRatio < 60) {
                    confirmed = false;
                    break;
                }
            }
            
            if(confirmed) {
                plateStartY = min(IMG_HEIGHT-1, y + EXPAND_MARGIN);
                Serial.printf("   → Biên trên tại Y=%d (tỷ lệ trắng: %.1f%%)\n", plateStartY, whiteRatio);
                break;
            }
        }
    }
    
    // VIỀN DƯỚI
    Serial.println("\n4. XÓA SÂU VIỀN DƯỚI:");
    plateEndY = IMG_HEIGHT - 1;
    for(int y = IMG_HEIGHT - 1; y >= 0; y -= SCAN_STEP) {
        int whiteCount = 0;
        int sampledPixels = 0;
        
        for(int x = plateStartX; x <= plateEndX; x += 2) {
            if(isWhitePixel(x, y)) whiteCount++;
            sampledPixels++;
        }
        
        float whiteRatio = (whiteCount * 100.0) / sampledPixels;
        
        if(whiteRatio >= REQUIRED_CONFIDENCE) {
            bool confirmed = true;
            for(int dy = y - 1; dy >= max(0, y - 3); dy--) {
                int nextWhiteCount = 0;
                for(int x = plateStartX; x <= plateEndX; x += 4) {
                    if(isWhitePixel(x, dy)) nextWhiteCount++;
                }
                float nextRatio = (nextWhiteCount * 100.0) / ((plateEndX - plateStartX + 1) / 4);
                if(nextRatio < 60) {
                    confirmed = false;
                    break;
                }
            }
            
            if(confirmed) {
                plateEndY = max(0, y - EXPAND_MARGIN);
                Serial.printf("   → Biên dưới tại Y=%d (tỷ lệ trắng: %.1f%%)\n", plateEndY, whiteRatio);
                break;
            }
        }
    }
    
    // KIỂM TRA TÍNH HỢP LỆ
    bool isValidCrop = (plateEndX - plateStartX > 50) && (plateEndY - plateStartY > 20);
    
    Serial.println("\n=== KẾT QUẢ XÓA VIỀN ===");
    Serial.printf("Vùng biển số: X[%d-%d] Y[%d-%d]\n", plateStartX, plateEndX, plateStartY, plateEndY);
    Serial.printf("Kích thước: %d × %d pixel\n", plateEndX-plateStartX+1, plateEndY-plateStartY+1);
    Serial.printf("Tính hợp lệ: %s\n", isValidCrop ? "HỢP LỆ" : "CÓ VẤN ĐỀ");
    
    if(!isValidCrop) {
        Serial.println("⚠️  Cảnh báo: Vùng crop không hợp lệ! Sử dụng vùng mặc định.");
        plateStartX = 10; plateStartY = 8;
        plateEndX = 280; plateEndY = 72;
    }
}

// =============================================
// PHÁT HIỆN VÙNG Y CHO TỪNG KÝ TỰ - PHIÊN BẢN CẢI TIẾN
// =============================================
std::pair<int, int> findCharacterVerticalBounds(int startX, int endX) {
    const int MIN_BLACK_PIXELS = 1; // GIẢM: chỉ cần 1 pixel đen là có ký tự
    const int MARGIN = 5; // TĂNG: margin lớn hơn
    
    int charStartY = plateEndY;
    int charEndY = plateStartY;
    bool foundBlack = false;
    
    // QUÉT THEO TỪNG DÒNG Y để tìm vùng thực sự có ký tự
    for(int y = plateStartY; y <= plateEndY; y++) {
        int blackCountInRow = 0;
        
        // Đếm số pixel đen trong dòng này của ký tự
        for(int x = startX; x <= endX; x++) {
            if(isBlackPixel(x, y)) {
                blackCountInRow++;
            }
        }
        
        // Nếu dòng này có ký tự
        if(blackCountInRow >= MIN_BLACK_PIXELS) {
            foundBlack = true;
            if(y < charStartY) charStartY = y;
            if(y > charEndY) charEndY = y;
        }
    }
    
    // Nếu không tìm thấy ký tự, dùng toàn bộ chiều cao
    if(!foundBlack) {
        charStartY = plateStartY;
        charEndY = plateEndY;
        Serial.printf("   ⚠️ Không tìm thấy ký tự, dùng toàn bộ chiều cao\n");
    } else {
        // THÊM MARGIN lớn hơn
        charStartY = max(plateStartY, charStartY - MARGIN);
        charEndY = min(plateEndY, charEndY + MARGIN);
        
        Serial.printf("   Vertical bounds: Y[%d-%d] height=%d\n", 
                     charStartY, charEndY, charEndY - charStartY + 1);
    }
    
    return {charStartY, charEndY};
}

// THÊM HÀM NÀY VÀO PHẦN RESIZE
bool resizeCharacterTo28x28(int startX, int endX, int charNum, ResizedCharacter& result) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int originalWidth = endX - startX + 1;
    int originalHeight = charEndY - charStartY + 1;
    
    if(originalWidth <= 0 || originalHeight <= 0) {
        return false;
    }
    
    result.originalWidth = originalWidth;
    result.originalHeight = originalHeight;
    
    // Clear background
    memset(result.pixels, 255, 784);
    
    float scaleX = (float)originalWidth / 28.0f;
    float scaleY = (float)originalHeight / 28.0f;
    
    // Nearest Neighbor (phương pháp cũ)
    for(int targetY = 0; targetY < 28; targetY++) {
        for(int targetX = 0; targetX < 28; targetX++) {
            int srcX = startX + (int)(targetX * scaleX);
            int srcY = charStartY + (int)(targetY * scaleY);
            
            srcX = max(startX, min(endX, srcX));
            srcY = max(charStartY, min(charEndY, srcY));
            
            result.pixels[targetY * 28 + targetX] = getPixel(srcX, srcY);
        }
    }
    
    Serial.printf("✅ Ký tự %d: %dx%d -> 28x28 (nearest)\n", 
                 charNum, originalWidth, originalHeight);
    return true;
}
// =============================================
// RESIZE KÝ TỰ VỚI BILINEAR INTERPOLATION
// =============================================
bool resizeCharacterTo28x28_Improved(int startX, int endX, int charNum, ResizedCharacter& result) {
    // Tìm giới hạn Y thực tế của ký tự này
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int originalWidth = endX - startX + 1;
    int originalHeight = charEndY - charStartY + 1;
    
    if(originalWidth <= 0 || originalHeight <= 0) {
        Serial.printf("❌ Ký tự %d: Kích thước không hợp lệ %dx%d\n", 
                     charNum, originalWidth, originalHeight);
        return false;
    }
    
    // KIỂM TRA: Nếu chiều cao quá nhỏ, dùng toàn bộ chiều cao biển số
    if(originalHeight < 20) {
        Serial.printf("   ⚠️ Chiều cao nhỏ (%d), dùng toàn bộ biển số\n", originalHeight);
        charStartY = plateStartY;
        charEndY = plateEndY;
        originalHeight = charEndY - charStartY + 1;
    }
    
    result.originalWidth = originalWidth;
    result.originalHeight = originalHeight;
    
    // Clear background trắng
    memset(result.pixels, 255, 784);
    
    // Tỷ lệ scale
    float scaleX = (float)originalWidth / 28.0f;
    float scaleY = (float)originalHeight / 28.0f;
    
    // Sử dụng Bilinear Interpolation để giữ chi tiết
    for(int targetY = 0; targetY < 28; targetY++) {
        for(int targetX = 0; targetX < 28; targetX++) {
            // Tính vị trí source với tọa độ thực
            float srcX = startX + (targetX * scaleX);
            float srcY = charStartY + (targetY * scaleY);
            
            // Lấy 4 pixel xung quanh để interpolate
            int x1 = (int)srcX;
            int y1 = (int)srcY;
            int x2 = min(endX, x1 + 1);
            int y2 = min(charEndY, y1 + 1);
            
            // Tính trọng số
            float wx = srcX - x1;
            float wy = srcY - y1;
            
            // Lấy 4 pixel xung quanh
            uint8_t p11 = getPixel(x1, y1);
            uint8_t p12 = getPixel(x1, y2);
            uint8_t p21 = getPixel(x2, y1);
            uint8_t p22 = getPixel(x2, y2);
            
            // Bilinear interpolation
            float interpolated = 
                p11 * (1 - wx) * (1 - wy) +
                p21 * wx * (1 - wy) +
                p12 * (1 - wx) * wy +
                p22 * wx * wy;
            
            result.pixels[targetY * 28 + targetX] = (uint8_t)interpolated;
        }
    }
    
    Serial.printf("✅ Ký tự %d: %dx%d -> 28x28 (bilinear)\n", 
                 charNum, originalWidth, originalHeight);
    return true;
}

// =============================================
// RESIZE VỚI SUPERSAMPLING (CHỐNG ALIASING)
// =============================================
bool resizeCharacterTo28x28_Supersampling(int startX, int endX, int charNum, ResizedCharacter& result) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int originalWidth = endX - startX + 1;
    int originalHeight = charEndY - charStartY + 1;
    
    if(originalWidth <= 0 || originalHeight <= 0) {
        return false;
    }
    
    result.originalWidth = originalWidth;
    result.originalHeight = originalHeight;
    
    // Clear background
    memset(result.pixels, 255, 784);
    
    float scaleX = (float)originalWidth / 28.0f;
    float scaleY = (float)originalHeight / 28.0f;
    
    // Supersampling 2x2
    const int SAMPLES = 2;
    
    for(int targetY = 0; targetY < 28; targetY++) {
        for(int targetX = 0; targetX < 28; targetX++) {
            float sum = 0;
            int sampleCount = 0;
            
            // Lấy nhiều mẫu trong 1 pixel đích
            for(int dy = 0; dy < SAMPLES; dy++) {
                for(int dx = 0; dx < SAMPLES; dx++) {
                    float sampleX = startX + (targetX + dx * 0.5f) * scaleX;
                    float sampleY = charStartY + (targetY + dy * 0.5f) * scaleY;
                    
                    int srcX = (int)sampleX;
                    int srcY = (int)sampleY;
                    
                    // Đảm bảo trong bounds
                    srcX = max(startX, min(endX, srcX));
                    srcY = max(charStartY, min(charEndY, srcY));
                    
                    sum += getPixel(srcX, srcY);
                    sampleCount++;
                }
            }
            
            result.pixels[targetY * 28 + targetX] = (uint8_t)(sum / sampleCount);
        }
    }
    
    Serial.printf("✅ Ký tự %d: %dx%d -> 28x28 (supersampling)\n", 
                 charNum, originalWidth, originalHeight);
    return true;
}

// RESIZE THÔNG MINH - TỰ ĐỘNG CHỌN PHƯƠNG PHÁP
// =============================================
bool resizeCharacterTo28x28_Smart(int startX, int endX, int charNum, ResizedCharacter& result) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;

    int originalHeight = charEndY - charStartY + 1;

    }

    float scaleY = (float)originalHeight / 28.0f;
    
    // Chọn phương pháp dựa trên kích thước gốc
    if(originalWidth >= 15 && originalHeight >= 15) {
        // Kích thước đủ lớn - dùng bilinear
        for(int targetY = 0; targetY < 28; targetY++) {
            for(int targetX = 0; targetX < 28; targetX++) {
                float srcX = startX + (targetX * scaleX);
                float srcY = charStartY + (targetY * scaleY);
                
                int x1 = (int)srcX;
                int y1 = (int)srcY;
                int x2 = min(endX, x1 + 1);
                int y2 = min(charEndY, y1 + 1);
                
                float wx = srcX - x1;
                float wy = srcY - y1;
                
                uint8_t p11 = getPixel(x1, y1);
                uint8_t p12 = getPixel(x1, y2);
                uint8_t p21 = getPixel(x2, y1);
                uint8_t p22 = getPixel(x2, y2);
                
                float interpolated = 
                    p11 * (1 - wx) * (1 - wy) +
                    p21 * wx * (1 - wy) +
                    p12 * (1 - wx) * wy +
                    p22 * wx * wy;
                
                result.pixels[targetY * 28 + targetX] = (uint8_t)interpolated;
            }
        }
        Serial.printf("✅ Ký tự %d: %dx%d -> 28x28 (bilinear)\n", originalWidth, originalHeight);
    } else {
        // Kích thước nhỏ - dùng nearest neighbor với threshold
        for(int targetY = 0; targetY < 28; targetY++) {
            for(int targetX = 0; targetX < 28; targetX++) {
                int srcX = startX + (int)(targetX * scaleX);
                int srcY = charStartY + (int)(targetY * scaleY);
                
                srcX = max(startX, min(endX, srcX));
                srcY = max(charStartY, min(charEndY, srcY));
                
                uint8_t pixel = getPixel(srcX, srcY);
                // Áp dụng threshold để làm rõ ký tự
                result.pixels[targetY * 28 + targetX] = pixel < 150 ? 0 : 255;
            }
        }
        Serial.printf("✅ Ký tự %d: %dx%d -> 28x28 (nearest+threshold)\n", originalWidth, originalHeight);
    }
    
    return true;
}

// =============================================
// RESIZE VỚI AREA AVERAGING (TỐT CHO DOWNSCALE LỚN)
// =============================================
bool resizeCharacterTo28x28_AreaAverage(int startX, int endX, int charNum, ResizedCharacter& result) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int originalWidth = endX - startX + 1;
    int originalHeight = charEndY - charStartY + 1;
    
    if(originalWidth <= 0 || originalHeight <= 0) {
        return false;
    }
    
    result.originalWidth = originalWidth;
    result.originalHeight = originalHeight;
    
    // Clear background trắng
    memset(result.pixels, 255, 784);
    
    // Tính kích thước cell cho area averaging
    float cellWidth = (float)originalWidth / 28.0f;
    float cellHeight = (float)originalHeight / 28.0f;
    
    for(int targetY = 0; targetY < 28; targetY++) {
        for(int targetX = 0; targetX < 28; targetX++) {
            // Tính vùng source tương ứng
            float srcStartX = startX + targetX * cellWidth;
            float srcStartY = charStartY + targetY * cellHeight;
            float srcEndX = srcStartX + cellWidth;
            float srcEndY = srcStartY + cellHeight;
            
            int startSrcX = (int)srcStartX;
            int startSrcY = (int)srcStartY;
            int endSrcX = min(endX, (int)srcEndX);
            int endSrcY = min(charEndY, (int)srcEndY);
            
            // Tính giá trị trung bình của vùng
            long sum = 0;
            int count = 0;
            
            for(int y = startSrcY; y <= endSrcY; y++) {
                for(int x = startSrcX; x <= endSrcX; x++) {
                    sum += getPixel(x, y);
                    count++;
                }
            }
            
            if(count > 0) {
                result.pixels[targetY * 28 + targetX] = (uint8_t)(sum / count);
            }
        }
    }
    
    Serial.printf("✅ Ký tự %d: %dx%d -> 28x28 (area average)\n", 
                 charNum, originalWidth, originalHeight);
    return true;
}

// =============================================
// RESIZE VỚI BILINEAR CẢI TIẾN (THÊM SHARPNESS)
// =============================================
bool resizeCharacterTo28x28_SharpBilinear(int startX, int endX, int charNum, ResizedCharacter& result) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int originalWidth = endX - startX + 1;
    int originalHeight = charEndY - charStartY + 1;
    
    if(originalWidth <= 0 || originalHeight <= 0) {
        return false;
    }
    
    result.originalWidth = originalWidth;
    result.originalHeight = originalHeight;
    
    memset(result.pixels, 255, 784);
    
    float scaleX = (float)originalWidth / 28.0f;
    float scaleY = (float)originalHeight / 28.0f;
    
    // Tăng sampling để giữ chi tiết
    const float SAMPLING_FACTOR = 2.0f;
    
    for(int targetY = 0; targetY < 28; targetY++) {
        for(int targetX = 0; targetX < 28; targetX++) {
            float centerX = startX + (targetX + 0.5f) * scaleX;
            float centerY = charStartY + (targetY + 0.5f) * scaleY;
            
            // Lấy 9 điểm xung quanh để tăng độ chi tiết
            float sum = 0;
            int samples = 0;
            
            for(float dy = -0.5f; dy <= 0.5f; dy += 0.5f) {
                for(float dx = -0.5f; dx <= 0.5f; dx += 0.5f) {
                    float sampleX = centerX + dx * SAMPLING_FACTOR;
                    float sampleY = centerY + dy * SAMPLING_FACTOR;
                    
                    int x1 = (int)sampleX;
                    int y1 = (int)sampleY;
                    int x2 = min(endX, x1 + 1);
                    int y2 = min(charEndY, y1 + 1);
                    
                    float wx = sampleX - x1;
                    float wy = sampleY - y1;
                    
                    uint8_t p11 = getPixel(x1, y1);
                    uint8_t p12 = getPixel(x1, y2);
                    uint8_t p21 = getPixel(x2, y1);
                    uint8_t p22 = getPixel(x2, y2);
                    
                    float interpolated = 
                        p11 * (1 - wx) * (1 - wy) +
                        p21 * wx * (1 - wy) +
                        p12 * (1 - wx) * wy +
                        p22 * wx * wy;
                    
                    sum += interpolated;
                    samples++;
                }
            }
            
            result.pixels[targetY * 28 + targetX] = (uint8_t)(sum / samples);
        }
    }
    
    Serial.printf("✅ Ký tự %d: %dx%d -> 28x28 (sharp bilinear)\n", 
                 charNum, originalWidth, originalHeight);
    return true;
}

// =============================================
// RESIZE VỚI ADAPTIVE METHOD (PHÂN VÙNG)
// =============================================
bool resizeCharacterTo28x28_Adaptive(int startX, int endX, int charNum, ResizedCharacter& result) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int originalWidth = endX - startX + 1;
    int originalHeight = charEndY - charStartY + 1;
    
    if(originalWidth <= 0 || originalHeight <= 0) {
        return false;
    }
    
    result.originalWidth = originalWidth;
    result.originalHeight = originalHeight;
    
    memset(result.pixels, 255, 784);
    
    // Phân tích độ phức tạp của ký tự
    bool isComplexCharacter = analyzeCharacterComplexity(startX, endX, charStartY, charEndY);
    
    if(isComplexCharacter && originalWidth > 20 && originalHeight > 40) {
        // Ký tự phức tạp và lớn - dùng area averaging
        return resizeCharacterTo28x28_AreaAverage(startX, endX, charNum, result);
    } else {
        // Ký tự đơn giản hoặc nhỏ - dùng sharp bilinear
        return resizeCharacterTo28x28_SharpBilinear(startX, endX, charNum, result);
    }
}

// =============================================
// PHÂN TÍCH ĐỘ PHỨC TẠP CỦA KÝ TỰ
// =============================================
bool analyzeCharacterComplexity(int startX, int endX, int startY, int endY) {
    int edgeCount = 0;
    int transitionCount = 0;
    
    // Đếm số transition (chuyển đổi trắng-đen)
    for(int y = startY; y <= endY; y++) {
        bool lastBlack = false;
        for(int x = startX; x <= endX; x++) {
            bool currentBlack = isBlackPixel(x, y);
            if(currentBlack != lastBlack) {
                transitionCount++;
                lastBlack = currentBlack;
            }
        }
    }
    
    // Tính độ phức tạp dựa trên số transition
    int totalPixels = (endX - startX + 1) * (endY - startY + 1);
    float complexity = (float)transitionCount / totalPixels;
    
    Serial.printf("   Độ phức tạp ký tự: %.3f (transitions: %d)\n", complexity, transitionCount);
    
    return complexity > 0.1f; // Ngưỡng độ phức tạp
}

// PHÂN TÁCH KÝ TỰ - THÊM LỌC NHIỄU
// =============================================
std::vector<std::pair<int, int>> findCharactersByVerticalProjection() {
    std::vector<std::pair<int, int>> characters;
    
    // Tính projection theo chiều ngang
    std::vector<int> horizontalProjection;
    for(int x = plateStartX; x <= plateEndX; x++) {
        int blackCount = 0;
        for(int y = plateStartY; y <= plateEndY; y++) {
            if(isBlackPixel(x, y)) blackCount++;
        }
        horizontalProjection.push_back(blackCount);
    }
    
    // Làm mịn projection để giảm nhiễu
    std::vector<int> smoothedProjection = horizontalProjection;
    for(int i = 1; i < horizontalProjection.size() - 1; i++) {
        smoothedProjection[i] = (horizontalProjection[i-1] + horizontalProjection[i] + horizontalProjection[i+1]) / 3;
    }
    
    // Tìm các vùng có ký tự
    const int MIN_VERTICAL_DENSITY = 2; // GIẢM ngưỡng
    const int MIN_CHAR_WIDTH = 5;
    const int MAX_CHAR_WIDTH = 40; // THÊM giới hạn trên
    
    int charStart = -1;
    bool inCharacter = false;
    
    for(int i = 0; i < smoothedProjection.size(); i++) {
        int x = plateStartX + i;
        
        if(smoothedProjection[i] >= MIN_VERTICAL_DENSITY) {
            if(!inCharacter) {
                charStart = x;
                inCharacter = true;
            }
        } else {
            if(inCharacter) {
                int charEnd = x - 1;
                int width = charEnd - charStart + 1;
                
                // KIỂM TRA kích thước hợp lý
                if(width >= MIN_CHAR_WIDTH && width <= MAX_CHAR_WIDTH) {
                    characters.push_back({charStart, charEnd});
                    Serial.printf("   Ký tự %d: X[%d-%d] width=%d\n", 
                                 characters.size(), charStart, charEnd, width);
                } else {
                    Serial.printf("   ⚠️ Bỏ qua vùng không hợp lệ: width=%d\n", width);
                }
                inCharacter = false;
            }
        }
    }
    
    // Xử lý ký tự cuối cùng
    if(inCharacter) {
        int charEnd = plateEndX;
        int width = charEnd - charStart + 1;
        if(width >= MIN_CHAR_WIDTH && width <= MAX_CHAR_WIDTH) {
            characters.push_back({charStart, charEnd});
        }
    }
    
    // LỌC KÝ TỰ THEO TỶ LỆ: loại bỏ ký tự quá hẹp/cao
    std::vector<std::pair<int, int>> filteredCharacters;
    for(auto& charRegion : characters) {
        int width = charRegion.second - charRegion.first + 1;
        auto verticalBounds = findCharacterVerticalBounds(charRegion.first, charRegion.second);
        int height = verticalBounds.second - verticalBounds.first + 1;
        
        float ratio = (float)width / height;
        
        // Tỷ lệ hợp lý cho ký tự (thường từ 0.3 đến 1.2)
        if(ratio >= 0.2 && ratio <= 1.5) {
            filteredCharacters.push_back(charRegion);
            Serial.printf("   ✅ Ký tự tỷ lệ: %.2f (w=%d, h=%d)\n", ratio, width, height);
        } else {
            Serial.printf("   ❌ Bỏ qua tỷ lệ không hợp lý: %.2f (w=%d, h=%d)\n", ratio, width, height);
        }
    }
    
    Serial.printf("Phát hiện: %d/%d ký tự hợp lệ\n", filteredCharacters.size(), characters.size());
    return filteredCharacters;
}

// HÀM TÍNH ĐỘ TƯƠNG PHẢN
// =============================================
float calculateContrast(const ResizedCharacter& character) {
    long sum = 0;
    long sumSquares = 0;
    int count = 784;
    
    for(int i = 0; i < 784; i++) {
        sum += character.pixels[i];
        sumSquares += character.pixels[i] * character.pixels[i];
    }
    
    float mean = (float)sum / count;
    float variance = (float)sumSquares / count - mean * mean;
    return sqrt(variance); // Độ lệch chuẩn = độ tương phản
}

struct PredictionResult {
    char predictedChar;
    float confidence;
    uint16_t labelIndex;
    bool success;
};

// Biến toàn cục để lưu input data cho callback
static uint8_t* current_image_data = nullptr;

// Hàm get_image_data theo cách pack RGB của bạn
static int get_image_data(size_t offset, size_t length, float *out_ptr) {
    if (current_image_data == nullptr) {
        return -1;
    }
    
    // Recalculate offset vào pixel index (per pixel, không per byte)
    size_t pixel_ix = offset;  // offset là pixel index (0 to 783)
    size_t pixels_left = length;
    size_t out_ptr_ix = 0;

    while (pixels_left != 0) {
        // Lấy giá trị grayscale từ current_image_data (R = G = B = gray)
        uint8_t gray = current_image_data[pixel_ix];  // gray từ 0-255

        // Pack giống RGB: r=g=b=gray, (gray << 16) + (gray << 8) + gray
        // Convert sang float như bạn đang làm
        out_ptr[out_ptr_ix] = (float)((gray << 16) + (gray << 8) + gray);

        out_ptr_ix++;
        pixel_ix++;
        pixels_left--;
    }
    return 0;
}


PredictionResult predictCharacterFromImage(uint8_t* image_data, int img_width, int img_height, bool debug_nn = false) {
    PredictionResult result;
    result.success = false;
    result.confidence = 0.0f;
    result.predictedChar = '?';
    result.labelIndex = 0;

    // Check kích thước khớp (model expect 784 features cho grayscale 28x28)
    if (EI_CLASSIFIER_RAW_SAMPLE_COUNT != img_width * img_height) {
        Serial.printf("❌ WARNING: Model features (%d) != Input image (%d)\n", 
                      EI_CLASSIFIER_RAW_SAMPLE_COUNT, img_width * img_height);
        return result;
    }

    // Kiểm tra dữ liệu đầu vào
    if (image_data == nullptr) {
        Serial.println("❌ ERR: Input image data is null");
        return result;
    }

    // Lưu con trỏ đến dữ liệu ảnh hiện tại
    current_image_data = image_data;

    // Chuẩn bị signal
    ei::signal_t signal;
    signal.total_length = EI_CLASSIFIER_RAW_SAMPLE_COUNT;
    signal.get_data = &get_image_data;

    ei_impulse_result_t ei_result = { 0 };

    // Chạy classifier
    EI_IMPULSE_ERROR res = run_classifier(&signal, &ei_result, debug_nn);
    if (res != EI_IMPULSE_OK) {
        Serial.printf("❌ ERR: Failed to run classifier (%d)\n", res);
        current_image_data = nullptr;
        return result;
    }

    // In thông tin timing
    Serial.printf("✅ Timing: DSP=%d ms, Class=%d ms", ei_result.timing.dsp, ei_result.timing.classification);
#if EI_CLASSIFIER_HAS_ANOMALY
    Serial.printf(", Anomaly=%d ms\n", ei_result.timing.anomaly);
#else
    Serial.println(" ms");
#endif

    // Tìm prediction có confidence cao nhất
    Serial.println("📊 Predictions:");
    float max_prob = 0.0f;
    uint16_t max_idx = 0;
    
    for (uint16_t i = 0; i < EI_CLASSIFIER_LABEL_COUNT; i++) {
        float prob = ei_result.classification[i].value;
        Serial.printf("  %s: %.5f\n", ei_classifier_inferencing_categories[i], prob);
        
        if (prob > max_prob) {
            max_prob = prob;
            max_idx = i;
        }
    }

    // Lấy ký tự dự đoán (giả sử categories là các ký tự như '0','1','2',...,'9','A','B',...)
    const char* predicted_label = ei_classifier_inferencing_categories[max_idx];
    
    // Nếu label có nhiều hơn 1 ký tự, lấy ký tự đầu tiên
    if (strlen(predicted_label) > 0) {
        result.predictedChar = predicted_label[0];
    } else {
        result.predictedChar = '?';
    }
    
    result.confidence = max_prob;
    result.labelIndex = max_idx;
    result.success = true;

    Serial.printf("🎯 Predicted: %s (%.5f)\n", predicted_label, max_prob);

#if EI_CLASSIFIER_HAS_ANOMALY
    Serial.printf("⚠️ Anomaly: %.3f\n", ei_result.anomaly);
    // Có thể thêm kiểm tra anomaly score nếu cần
    if (ei_result.anomaly > 0.5) { // Ngưỡng anomaly, điều chỉnh theo model
        Serial.println("🚨 High anomaly detected - prediction may be unreliable");
    }
#endif

    current_image_data = nullptr;
    return result;
}

void setup() {
    Serial.begin(115200);
    delay(1000);
    
    Serial.println("=== KHỞI ĐỘNG HỆ THỐNG ===");
    
    // KIỂM TRA PSRAM
    if(!psramFound()){
        Serial.println("❌ LỖI: PSRAM không tìm thấy!");
        return;
    }
    
    Serial.printf("✅ PSRAM: %d bytes\n", ESP.getPsramSize());
    Serial.printf("✅ Free PSRAM: %d bytes\n", ESP.getFreePsram());
    
    // CẤP PHÁT PSRAM
    psram_img = (uint8_t*)ps_malloc(IMG_WIDTH * IMG_HEIGHT);
    if(psram_img == nullptr) {
        Serial.println("❌ LỖI: Không thể cấp phát PSRAM!");
        return;
    }
    
    // COPY DỮ LIỆU
    if(memcpy_P(psram_img, input_img, IMG_WIDTH * IMG_HEIGHT) == nullptr) {
        Serial.println("❌ LỖI: Copy dữ liệu thất bại!");
        free(psram_img);
        return;
    }
    
    Serial.println("✅ Image loaded to PSRAM successfully");
    Serial.printf("✅ Image size: %d x %d = %d bytes\n", IMG_WIDTH, IMG_HEIGHT, IMG_WIDTH * IMG_HEIGHT);
    
}
