uint8_t getPixel(int x, int y) {
    if(x < 0 || x >= IMG_WIDTH || y < 0 || y >= IMG_HEIGHT) return 0;
    return psram_img[y * IMG_WIDTH + x];
}

bool isBlackPixel(int x, int y) {
    return getPixel(x, y) < 97;
}

bool isWhitePixel(int x, int y) {
    return getPixel(x, y) > 130;
}

int min(int a, int b) { return (a < b) ? a : b; }
int max(int a, int b) { return (a > b) ? a : b; }

// =============================================
// CROP VIỀN THỰC TẾ (giữ nguyên)
// =============================================
void deepCropBorder() {
    Serial.println("\n=== XÓA VIỀN SÂU ===");
    
    const int REQUIRED_CONFIDENCE = 70;
    const int SCAN_STEP = 2;
    const int EXPAND_MARGIN = 3;
    
    // VIỀN TRÁI
    plateStartX = 0;
    for(int x = 0; x < IMG_WIDTH; x += SCAN_STEP) {
        int whiteCount = 0;
        int sampledPixels = 0;
        
        for(int y = 0; y < IMG_HEIGHT; y += 2) {
            if(isWhitePixel(x, y)) whiteCount++;
            sampledPixels++;
        }
        
        float whiteRatio = (whiteCount * 100.0) / sampledPixels;
        
        if(whiteRatio >= REQUIRED_CONFIDENCE) {
            bool confirmed = true;
            for(int dx = x + 1; dx <= min(IMG_WIDTH-1, x + 3); dx++) {
                int nextWhiteCount = 0;
                for(int y = 0; y < IMG_HEIGHT; y += 4) {
                    if(isWhitePixel(dx, y)) nextWhiteCount++;
                }
                float nextRatio = (nextWhiteCount * 100.0) / (IMG_HEIGHT / 4);
                if(nextRatio < 60) {
                    confirmed = false;
                    break;
                }
            }
            
            if(confirmed) {
                plateStartX = min(IMG_WIDTH-1, x + EXPAND_MARGIN);
                break;
            }
        }
    }
    
    // VIỀN PHẢI
    plateEndX = IMG_WIDTH - 1;
    for(int x = IMG_WIDTH - 1; x >= 0; x -= SCAN_STEP) {
        int whiteCount = 0;
        int sampledPixels = 0;
        
        for(int y = 0; y < IMG_HEIGHT; y += 2) {
            if(isWhitePixel(x, y)) whiteCount++;
            sampledPixels++;
        }
        
        float whiteRatio = (whiteCount * 100.0) / sampledPixels;
        
        if(whiteRatio >= REQUIRED_CONFIDENCE) {
            bool confirmed = true;
            for(int dx = x - 1; dx >= max(0, x - 3); dx--) {
                int nextWhiteCount = 0;
                for(int y = 0; y < IMG_HEIGHT; y += 4) {
                    if(isWhitePixel(dx, y)) nextWhiteCount++;
                }
                float nextRatio = (nextWhiteCount * 100.0) / (IMG_HEIGHT / 4);
                if(nextRatio < 60) {
                    confirmed = false;
                    break;
                }
            }
            
            if(confirmed) {
                plateEndX = max(0, x - EXPAND_MARGIN);
                break;
            }
        }
    }
    
    // VIỀN TRÊN
    plateStartY = 0;
    for(int y = 0; y < IMG_HEIGHT; y += SCAN_STEP) {
        int whiteCount = 0;
        int sampledPixels = 0;
        
        for(int x = plateStartX; x <= plateEndX; x += 2) {
            if(isWhitePixel(x, y)) whiteCount++;
            sampledPixels++;
        }
        
        float whiteRatio = (whiteCount * 100.0) / sampledPixels;
        
        if(whiteRatio >= REQUIRED_CONFIDENCE) {
            bool confirmed = true;
            for(int dy = y + 1; dy <= min(IMG_HEIGHT-1, y + 3); dy++) {
                int nextWhiteCount = 0;
                for(int x = plateStartX; x <= plateEndX; x += 4) {
                    if(isWhitePixel(x, dy)) nextWhiteCount++;
                }
                float nextRatio = (nextWhiteCount * 100.0) / ((plateEndX - plateStartX + 1) / 4);
                if(nextRatio < 60) {
                    confirmed = false;
                    break;
                }
            }
            
            if(confirmed) {
                plateStartY = min(IMG_HEIGHT-1, y + EXPAND_MARGIN);
                break;
            }
        }
    }
    
    // VIỀN DƯỚI
    plateEndY = IMG_HEIGHT - 1;
    for(int y = IMG_HEIGHT - 1; y >= 0; y -= SCAN_STEP) {
        int whiteCount = 0;
        int sampledPixels = 0;
        
        for(int x = plateStartX; x <= plateEndX; x += 2) {
            if(isWhitePixel(x, y)) whiteCount++;
            sampledPixels++;
        }
        
        float whiteRatio = (whiteCount * 100.0) / sampledPixels;
        
        if(whiteRatio >= REQUIRED_CONFIDENCE) {
            bool confirmed = true;
            for(int dy = y - 1; dy >= max(0, y - 3); dy--) {
                int nextWhiteCount = 0;
                for(int x = plateStartX; x <= plateEndX; x += 4) {
                    if(isWhitePixel(x, dy)) nextWhiteCount++;
                }
                float nextRatio = (nextWhiteCount * 100.0) / ((plateEndX - plateStartX + 1) / 4);
                if(nextRatio < 60) {
                    confirmed = false;
                    break;
                }
            }
            
            if(confirmed) {
                plateEndY = max(0, y - EXPAND_MARGIN);
                break;
            }
        }
    }
    
    bool isValidCrop = (plateEndX - plateStartX > 50) && (plateEndY - plateStartY > 20);
    
    if(!isValidCrop) {
        plateStartX = 10; plateStartY = 8;
        plateEndX = 280; plateEndY = 72;
    }
}

// =============================================
// PHÁT HIỆN VÙNG Y CHO TỪNG KÝ TỰ (giữ nguyên)
// =============================================
std::pair<int, int> findCharacterVerticalBounds(int startX, int endX) {
    const int MIN_BLACK_PIXELS = 1;
    const int MARGIN = 5;
    
    int charStartY = plateEndY;
    int charEndY = plateStartY;
    bool foundBlack = false;
    
    for(int y = plateStartY; y <= plateEndY; y++) {
        int blackCountInRow = 0;
        
        for(int x = startX; x <= endX; x++) {
            if(isBlackPixel(x, y)) {
                blackCountInRow++;
            }
        }
        
        if(blackCountInRow >= MIN_BLACK_PIXELS) {
            foundBlack = true;
            if(y < charStartY) charStartY = y;
            if(y > charEndY) charEndY = y;
        }
    }
    
    if(!foundBlack) {
        charStartY = plateStartY;
        charEndY = plateEndY;
    } else {
        charStartY = max(plateStartY, charStartY - MARGIN);
        charEndY = min(plateEndY, charEndY + MARGIN);
    }
    
    return {charStartY, charEndY};
}

// CÁC PHƯƠNG PHÁP RESIZE (giữ nguyên)
// =============================================
bool resizeCharacterTo28x28_Nearest(int startX, int endX, ResizedCharacter& result) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int originalWidth = endX - startX + 1;
    int originalHeight = charEndY - charStartY + 1;
    
    if(originalWidth <= 0 || originalHeight <= 0) {
        return false;
    }
    
    result.originalWidth = originalWidth;
    result.originalHeight = originalHeight;
    result.methodName = "Nearest";
    
    memset(result.pixels, 255, 784);
    
    float scaleX = (float)originalWidth / 28.0f;
    float scaleY = (float)originalHeight / 28.0f;
    
    for(int targetY = 0; targetY < 28; targetY++) {
        for(int targetX = 0; targetX < 28; targetX++) {
            int srcX = startX + (int)(targetX * scaleX);
            int srcY = charStartY + (int)(targetY * scaleY);
            
            srcX = max(startX, min(endX, srcX));
            srcY = max(charStartY, min(charEndY, srcY));
            
            result.pixels[targetY * 28 + targetX] = getPixel(srcX, srcY);
        }
    }
    
    return true;
}

bool resizeCharacterTo28x28_Bilinear(int startX, int endX, ResizedCharacter& result) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int originalWidth = endX - startX + 1;
    int originalHeight = charEndY - charStartY + 1;
    
    if(originalWidth <= 0 || originalHeight <= 0) {
        return false;
    }
    
    result.originalWidth = originalWidth;
    result.originalHeight = originalHeight;
    result.methodName = "Bilinear";
    
    memset(result.pixels, 255, 784);
    
    float scaleX = (float)originalWidth / 28.0f;
    float scaleY = (float)originalHeight / 28.0f;
    
    for(int targetY = 0; targetY < 28; targetY++) {
        for(int targetX = 0; targetX < 28; targetX++) {
            float srcX = startX + (targetX * scaleX);
            float srcY = charStartY + (targetY * scaleY);
            
            int x1 = (int)srcX;
            int y1 = (int)srcY;
            int x2 = min(endX, x1 + 1);
            int y2 = min(charEndY, y1 + 1);
            
            float wx = srcX - x1;
            float wy = srcY - y1;
            
            uint8_t p11 = getPixel(x1, y1);
            uint8_t p12 = getPixel(x1, y2);
            uint8_t p21 = getPixel(x2, y1);
            uint8_t p22 = getPixel(x2, y2);
            
            float interpolated = 
                p11 * (1 - wx) * (1 - wy) +
                p21 * wx * (1 - wy) +
                p12 * (1 - wx) * wy +
                p22 * wx * wy;
            
            result.pixels[targetY * 28 + targetX] = (uint8_t)interpolated;
        }
    }
    
    return true;
}

bool resizeCharacterTo28x28_AreaAverage(int startX, int endX, ResizedCharacter& result) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int originalWidth = endX - startX + 1;
    int originalHeight = charEndY - charStartY + 1;
    
    if(originalWidth <= 0 || originalHeight <= 0) {
        return false;
    }
    
    result.originalWidth = originalWidth;
    result.originalHeight = originalHeight;
    result.methodName = "AreaAverage";
    
    memset(result.pixels, 255, 784);
    
    float cellWidth = (float)originalWidth / 28.0f;
    float cellHeight = (float)originalHeight / 28.0f;
    
    for(int targetY = 0; targetY < 28; targetY++) {
        for(int targetX = 0; targetX < 28; targetX++) {
            float srcStartX = startX + targetX * cellWidth;
            float srcStartY = charStartY + targetY * cellHeight;
            float srcEndX = srcStartX + cellWidth;
            float srcEndY = srcStartY + cellHeight;
            
            int startSrcX = (int)srcStartX;
            int startSrcY = (int)srcStartY;
            int endSrcX = min(endX, (int)srcEndX);
            int endSrcY = min(charEndY, (int)srcEndY);
            
            long sum = 0;
            int count = 0;
            
            for(int y = startSrcY; y <= endSrcY; y++) {
                for(int x = startSrcX; x <= endSrcX; x++) {
                    sum += getPixel(x, y);
                    count++;
                }
            }
            
            if(count > 0) {
                result.pixels[targetY * 28 + targetX] = (uint8_t)(sum / count);
            }
        }
    }
    
    return true;
}

// HÀM KIỂM TRA KÍCH THƯỚC KÝ TỰ HỢP LỆ

bool isValidCharacterSize(int startX, int endX) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int width = endX - startX + 1;
    int height = charEndY - charStartY + 1;
    
    // Ngưỡng kích thước tối thiểu
    const int MIN_WIDTH = 20;   // Chiều rộng tối thiểu
    const int MIN_HEIGHT = 30;  // Chiều cao tối thiểu
    const int MAX_WIDTH = 80;  // Chiều rộng tối đa (tránh ký tự quá lớn)
    const int MAX_HEIGHT = 100;  // Chiều cao tối đa
    
    bool isValid = (width >= MIN_WIDTH) && (height >= MIN_HEIGHT) && 
                   (width <= MAX_WIDTH) && (height <= MAX_HEIGHT);
    
    if (!isValid) {
        Serial.printf("   ❌ Bỏ qua ký tự: %dx%d (vượt ngưỡng %dx%d - %dx%d)\n", 
                     width, height, MIN_WIDTH, MIN_HEIGHT, MAX_WIDTH, MAX_HEIGHT);
    } else {
        Serial.printf("   ✅ Ký tự hợp lệ: %dx%d\n", width, height);
    }
    
    return isValid;
}

// =============================================
// HÀM KIỂM TRA TỶ LỆ KÝ TỰ HỢP LỆ
// =============================================
bool isValidCharacterRatio(int startX, int endX) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int width = endX - startX + 1;
    int height = charEndY - charStartY + 1;
    
    float ratio = (float)width / height;
    
    // Tỷ lệ hợp lý cho ký tự biển số (thường từ 0.3 đến 1.2)
    const float MIN_RATIO = 0.3f;
    const float MAX_RATIO = 1.5f;
    
    bool isValid = (ratio >= MIN_RATIO) && (ratio <= MAX_RATIO);
    
    if (!isValid) {
        Serial.printf("   ❌ Bỏ qua tỷ lệ: %.2f (vượt ngưỡng %.1f-%.1f)\n", 
                     ratio, MIN_RATIO, MAX_RATIO);
    } else {
        Serial.printf("   ✅ Tỷ lệ hợp lệ: %.2f\n", ratio);
    }
    
    return isValid;
}

// =============================================
// HÀM KIỂM TRA MẬT ĐỘ PIXEL ĐEN
// =============================================
bool hasSufficientBlackPixels(int startX, int endX) {
    auto verticalBounds = findCharacterVerticalBounds(startX, endX);
    int charStartY = verticalBounds.first;
    int charEndY = verticalBounds.second;
    
    int totalPixels = (endX - startX + 1) * (charEndY - charStartY + 1);
    if (totalPixels == 0) return false;
    
    int blackPixelCount = 0;
    for (int y = charStartY; y <= charEndY; y++) {
        for (int x = startX; x <= endX; x++) {
            if (isBlackPixel(x, y)) {
                blackPixelCount++;
            }
        }
    }
    
    float blackDensity = (float)blackPixelCount / totalPixels * 100.0f;
    
    // Ngưỡng mật độ pixel đen tối thiểu (tránh nhiễu)
    const float MIN_BLACK_DENSITY = 1.0f; // 1%
    const float MAX_BLACK_DENSITY = 70.0f; // 70% (tránh vùng toàn đen)
    
    bool isValid = (blackDensity >= MIN_BLACK_DENSITY) && (blackDensity <= MAX_BLACK_DENSITY);
    
    if (!isValid) {
        Serial.printf("   ❌ Mật độ đen: %.1f%% (vượt ngưỡng %.1f-%.1f%%)\n", 
                     blackDensity, MIN_BLACK_DENSITY, MAX_BLACK_DENSITY);
    } else {
        Serial.printf("   ✅ Mật độ đen: %.1f%%\n", blackDensity);
    }
    
    return isValid;
}

// =============================================
// HÀM KIỂM TRA TOÀN DIỆN KÝ TỰ
// =============================================
bool isValidCharacter(int startX, int endX) {
    Serial.printf("   🔍 Kiểm tra ký tự X[%d-%d]:\n", startX, endX);
    
    bool validSize = isValidCharacterSize(startX, endX);
    bool validRatio = isValidCharacterRatio(startX, endX);
    bool validDensity = hasSufficientBlackPixels(startX, endX);
    
    bool isValid = validSize && validRatio && validDensity;
    
    if (isValid) {
        Serial.printf("   ✅ Ký tự HOÀN TOÀN HỢP LỆ\n");
    } else {
        Serial.printf("   ❌ Ký tự KHÔNG HỢP LỆ\n");
    }
    
    return isValid;
}

// CẬP NHẬT HÀM PHÂN TÁCH KÝ TỰ (CÓ LỌC)
// =============================================
std::vector<std::pair<int, int>> findCharactersByVerticalProjectionWithFilter() {
    std::vector<std::pair<int, int>> characters;
    
    std::vector<int> horizontalProjection;
    for(int x = plateStartX; x <= plateEndX; x++) {
        int blackCount = 0;
        for(int y = plateStartY; y <= plateEndY; y++) {
            if(isBlackPixel(x, y)) blackCount++;
        }
        horizontalProjection.push_back(blackCount);
    }
    
    std::vector<int> smoothedProjection = horizontalProjection;
    for(int i = 1; i < horizontalProjection.size() - 1; i++) {
        smoothedProjection[i] = (horizontalProjection[i-1] + horizontalProjection[i] + horizontalProjection[i+1]) / 3;
    }
    
    const int MIN_VERTICAL_DENSITY = 2;
    const int MIN_CHAR_WIDTH = 5;
    const int MAX_CHAR_WIDTH = 100; // Tăng lên để lọc sau
    
    int charStart = -1;
    bool inCharacter = false;
    
    for(int i = 0; i < smoothedProjection.size(); i++) {
        int x = plateStartX + i;
        
        if(smoothedProjection[i] >= MIN_VERTICAL_DENSITY) {
            if(!inCharacter) {
                charStart = x;
                inCharacter = true;
            }
        } else {
            if(inCharacter) {
                int charEnd = x - 1;
                int width = charEnd - charStart + 1;
                
                if(width >= MIN_CHAR_WIDTH && width <= MAX_CHAR_WIDTH) {
                    characters.push_back({charStart, charEnd});
                }
                inCharacter = false;
            }
        }
    }
    
    if(inCharacter) {
        int charEnd = plateEndX;
        int width = charEnd - charStart + 1;
        if(width >= MIN_CHAR_WIDTH && width <= MAX_CHAR_WIDTH) {
            characters.push_back({charStart, charEnd});
        }
    }
    
    // LỌC KÝ TỰ THEO TIÊU CHÍ MỚI
    std::vector<std::pair<int, int>> filteredCharacters;
    Serial.printf("\n=== LỌC KÝ TỰ TỰ ĐỘNG ===\n");
    Serial.printf("Phát hiện ban đầu: %d ký tự\n", characters.size());
    
    for(auto& charRegion : characters) {
        if(isValidCharacter(charRegion.first, charRegion.second)) {
            filteredCharacters.push_back(charRegion);
        }
    }
    
    Serial.printf("Sau khi lọc: %d ký tự hợp lệ\n", filteredCharacters.size());
    
    return filteredCharacters;
}

static uint8_t* current_image_data = nullptr;

static int get_image_data(size_t offset, size_t length, float *out_ptr) {
    if (current_image_data == nullptr) {
        return -1;
    }
    
    size_t pixel_ix = offset;
    size_t pixels_left = length;
    size_t out_ptr_ix = 0;

    while (pixels_left != 0) {
        uint8_t gray = current_image_data[pixel_ix];
        out_ptr[out_ptr_ix] = (float)((gray << 16) + (gray << 8) + gray);

        out_ptr_ix++;
        pixel_ix++;
        pixels_left--;
    }
    return 0;
}

PredictionResult predictCharacterFromImage(uint8_t* image_data, int img_width, int img_height, const String& methodName, bool debug_nn = false) {
    PredictionResult result;
    result.success = false;
    result.confidence = 0.0f;
    result.predictedChar = '?';
    result.labelIndex = 0;
    result.methodName = methodName;

    if (EI_CLASSIFIER_RAW_SAMPLE_COUNT != img_width * img_height) {
        return result;
    }

    if (image_data == nullptr) {
        return result;
    }

    current_image_data = image_data;

    ei::signal_t signal;
    signal.total_length = EI_CLASSIFIER_RAW_SAMPLE_COUNT;
    signal.get_data = &get_image_data;

    ei_impulse_result_t ei_result = { 0 };

    EI_IMPULSE_ERROR res = run_classifier(&signal, &ei_result, debug_nn);
    if (res != EI_IMPULSE_OK) {
        current_image_data = nullptr;
        return result;
    }

    float max_prob = 0.0f;
    uint16_t max_idx = 0;
    
    for (uint16_t i = 0; i < EI_CLASSIFIER_LABEL_COUNT; i++) {
        float prob = ei_result.classification[i].value;
        if (prob > max_prob) {
            max_prob = prob;
            max_idx = i;
        }
    }

    const char* predicted_label = ei_classifier_inferencing_categories[max_idx];
    
    if (strlen(predicted_label) > 0) {
        result.predictedChar = predicted_label[0];
    } else {
        result.predictedChar = '?';
    }
    
    result.confidence = max_prob;
    result.labelIndex = max_idx;
    result.success = true;

    current_image_data = nullptr;
    return result;
}

PredictionResult predictResizedCharacter(const ResizedCharacter& character) {
    return predictCharacterFromImage((uint8_t*)character.pixels, 28, 28, character.methodName);
}

// HÀM HIỂN THỊ KÝ TỰ RESIZE
// =============================================
void displayCharacterStatistics(const std::vector<std::pair<int, int>>& characters) {
    Serial.println("\n=== THỐNG KÊ KÝ TỰ PHÁT HIỆN ===");
    
    for(int i = 0; i < characters.size(); i++) {
        auto region = characters[i];
        auto verticalBounds = findCharacterVerticalBounds(region.first, region.second);
        
        int width = region.second - region.first + 1;
        int height = verticalBounds.second - verticalBounds.first + 1;
        float ratio = (float)width / height;
        
        // Tính mật độ pixel đen
        int blackPixels = 0;
        int totalPixels = width * height;
        for(int y = verticalBounds.first; y <= verticalBounds.second; y++) {
            for(int x = region.first; x <= region.second; x++) {
                if(isBlackPixel(x, y)) blackPixels++;
            }
        }
        float blackDensity = (float)blackPixels / totalPixels * 100.0f;
        
        Serial.printf("Ký tự %d: %dx%d (tỷ lệ: %.2f, đen: %.1f%%) X[%d-%d] Y[%d-%d]\n",
                     i+1, width, height, ratio, blackDensity,
                     region.first, region.second, verticalBounds.first, verticalBounds.second);
    }
}

void processAndSaveAllCharacters() {
    Serial.println("\n=== XỬ LÝ VÀ LƯU TOÀN BỘ KÝ TỰ (CÓ LỌC) ===");
    
    // Reset dữ liệu cũ
    allCharactersData.clear();
    
    // Bước 1: Xóa viền
    deepCropBorder();
    
    // Bước 2: Phân tách ký tự VÀ LỌC
    std::vector<std::pair<int, int>> characters = findCharactersByVerticalProjectionWithFilter();
    
    // Hiển thị thống kê
    displayCharacterStatistics(characters);
    
    if(characters.empty()) {
        Serial.println("⚠️  Cảnh báo: Không tìm thấy ký tự hợp lệ nào!");
        return;
    }
    
    // Bước 3: Xử lý từng ký tự hợp lệ với tất cả phương pháp
    for(int i = 0; i < min(10, (int)characters.size()); i++) {
        auto region = characters[i];
        
        CharacterData charData;
        charData.characterIndex = i + 1;
        charData.startX = region.first;
        charData.endX = region.second;
        
        Serial.printf("\n🧪 Xử lý ký tự %d: X[%d-%d]\n", i+1, region.first, region.second);
        
        // Hiển thị thông tin kích thước
        auto verticalBounds = findCharacterVerticalBounds(region.first, region.second);
        Serial.printf("   Kích thước: %dx%d\n", 
                     region.second - region.first + 1,
                     verticalBounds.second - verticalBounds.first + 1);
        
        // Danh sách các phương pháp resize
        std::vector<std::pair<String, bool(*)(int, int, ResizedCharacter&)>> methods = {
            {"Nearest", resizeCharacterTo28x28_Nearest},
            {"Bilinear", resizeCharacterTo28x28_Bilinear},
            {"AreaAverage", resizeCharacterTo28x28_AreaAverage}
        };
        
        // Áp dụng tất cả phương pháp
        for(auto& method : methods) {
            ResizedCharacter resizedChar;
            bool success = method.second(region.first, region.second, resizedChar);
            
            if(success) {
                // Lưu ảnh đã resize
                charData.resizedVersions.push_back(resizedChar);
                
                // Dự đoán và lưu kết quả
                PredictionResult predResult = predictResizedCharacter(resizedChar);
                charData.predictions.push_back(predResult);
                
                Serial.printf("  ✅ %s: %c (%.3f)\n", 
                             method.first.c_str(), predResult.predictedChar, predResult.confidence);
            } else {
                Serial.printf("  ❌ %s: Lỗi resize\n", method.first.c_str());
            }
        }
        
        // Lưu dữ liệu ký tự
        allCharactersData.push_back(charData);
    }
    
    // Serial.printf("\n✅ Đã lưu toàn bộ %d ký tự hợp lệ với %d phương pháp resize\n", 
    //              allCharactersData.size(), methods.size());
}


// =============================================
// HÀM HIỂN THỊ BẢNG SO SÁNH CHI TIẾT
// =============================================
void displayDetailedComparisonTable() {
    Serial.println("\n" + String('=') + String(100, '='));
    Serial.println("📊 BẢNG SO SÁNH CHI TIẾT TOÀN BỘ KÝ TỰ");
    Serial.println(String('=') + String(100, '='));
    
    if(allCharactersData.empty()) {
        Serial.println("❌ Không có dữ liệu để so sánh");
        return;
    }
    
    // Header
    Serial.printf("%-8s %-12s", "Ký tự", "Kích thước");
    std::vector<String> methods = {"Nearest", "Bilinear", "AreaAverage"};
    for(auto& method : methods) {
        Serial.printf(" | %-20s", method.c_str());
    }
    Serial.println();
    Serial.println(String('-') + String(100, '-'));
    
    // Data rows
    for(auto& charData : allCharactersData) {
        // Hiển thị thông tin ký tự
        auto verticalBounds = findCharacterVerticalBounds(charData.startX, charData.endX);
        int height = verticalBounds.second - verticalBounds.first + 1;
        int width = charData.endX - charData.startX + 1;
        
        Serial.printf("%-8d %-12s", charData.characterIndex, (String(width) + "x" + String(height)).c_str());
        
        // Hiển thị kết quả cho từng phương pháp
        for(auto& method : methods) {
            bool found = false;
            for(int i = 0; i < charData.predictions.size(); i++) {
                if(charData.predictions[i].methodName == method && charData.predictions[i].success) {
                    Serial.printf(" | %c (%.3f) %dx%d", 
                                 charData.predictions[i].predictedChar,
                                 charData.predictions[i].confidence,
                                 charData.resizedVersions[i].originalWidth,
                                 charData.resizedVersions[i].originalHeight);
                    found = true;
                    break;
                }
            }
            if(!found) {
                Serial.printf(" | %-20s", "N/A");
            }
        }
        Serial.println();
    }
    
    // Thống kê tổng quan
    Serial.println(String('-') + String(100, '-'));
    Serial.println("📈 THỐNG KÊ TỔNG QUAN:");
    
    for(auto& method : methods) {
        int totalChars = 0;
        float totalConfidence = 0.0f;
        std::vector<char> predictedChars;
        
        for(auto& charData : allCharactersData) {
            for(auto& prediction : charData.predictions) {
                if(prediction.methodName == method && prediction.success) {
                    totalChars++;
                    totalConfidence += prediction.confidence;
                    predictedChars.push_back(prediction.predictedChar);
                }
            }
        }
        
        if(totalChars > 0) {
            float avgConfidence = totalConfidence / totalChars;
            
            // Hiển thị các ký tự đã dự đoán
            String charsStr = "";
            for(char c : predictedChars) {
                charsStr += String(c) + " ";
            }
            
            Serial.printf("  %-12s: %d ký tự, Độ tin cậy TB: %.3f\n", 
                         method.c_str(), totalChars, avgConfidence);
            Serial.printf("                Ký tự: %s\n", charsStr.c_str());
        }
    }
    
    // So sánh sự khác biệt giữa các phương pháp
    Serial.println("\n🔍 SO SÁNH SỰ KHÁC BIỆT:");
    for(auto& charData : allCharactersData) {
        if(charData.predictions.size() >= 2) {
            bool allSame = true;
            char firstChar = charData.predictions[0].predictedChar;
            
            for(int i = 1; i < charData.predictions.size(); i++) {
                if(charData.predictions[i].predictedChar != firstChar) {
                    allSame = false;
                    break;
                }
            }
            
            if(!allSame) {
                Serial.printf("  Ký tự %d: ", charData.characterIndex);
                for(auto& pred : charData.predictions) {
                    if(pred.success) {
                        Serial.printf("%s=%c(%.3f) ", pred.methodName.c_str(), pred.predictedChar, pred.confidence);
                    }
                }
                Serial.println();
            }
        }
    }
    
    Serial.println(String('=') + String(100, '='));
}

// HÀM TRUY XUẤT KÝ TỰ CỤ THỂ
// =============================================
void displayCharacterDetails(int charIndex) {
    if(charIndex < 1 || charIndex > allCharactersData.size()) {
        Serial.printf("❌ Ký tự %d không tồn tại\n", charIndex);
        return;
    }
    
    CharacterData& charData = allCharactersData[charIndex-1];
    
    Serial.printf("\n🔍 CHI TIẾT KÝ TỰ %d:\n", charIndex);
    Serial.printf("Vị trí: X[%d-%d]\n", charData.startX, charData.endX);
    
    auto verticalBounds = findCharacterVerticalBounds(charData.startX, charData.endX);
    Serial.printf("Kích thước gốc: %dx%d\n", 
                 charData.endX - charData.startX + 1,
                 verticalBounds.second - verticalBounds.first + 1);
    
    Serial.println("Kết quả dự đoán:");
    for(int i = 0; i < charData.predictions.size(); i++) {
        if(charData.predictions[i].success) {
            Serial.printf("  %-12s: %c (độ tin cậy: %.3f)\n",
                         charData.predictions[i].methodName.c_str(),
                         charData.predictions[i].predictedChar,
                         charData.predictions[i].confidence);
        }
    }
}

// HÀM TÌM PHƯƠNG PHÁP TỐT NHẤT CHO TỪNG KÝ TỰ
// =============================================
void findBestMethods() {
    Serial.println("\n🏆 PHƯƠNG PHÁP TỐT NHẤT CHO TỪNG KÝ TỰ:");
    
    for(auto& charData : allCharactersData) {
        PredictionResult bestPred;
        bestPred.confidence = 0.0f;
        bestPred.success = false;
        
        for(auto& pred : charData.predictions) {
            if(pred.success && pred.confidence > bestPred.confidence) {
                bestPred = pred;
            }
        }
        
        if(bestPred.success) {
            Serial.printf("  Ký tự %d: %s với %c (%.3f)\n",
                         charData.characterIndex,
                         bestPred.methodName.c_str(),
                         bestPred.predictedChar,
                         bestPred.confidence);
        }
    }
}

        return;
    }

    if(psram_img == nullptr) {
        Serial.println("❌ LỖI: Không thể cấp phát PSRAM!");
        return;
    }
    
    if(memcpy_P(psram_img, input_img, IMG_WIDTH * IMG_HEIGHT) == nullptr) {
        Serial.println("❌ LỖI: Copy dữ liệu thất bại!");
        free(psram_img);
        return;
    }
    
    Serial.println("✅ Image loaded to PSRAM successfully");
}
